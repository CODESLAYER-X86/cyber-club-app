   443→  /* ─── Enhanced Alignment Guides ─── */
   444→
   445→  const snapToGuides = (x: number, y: number, elementWidth?: number, elementHeight?: number) => {
   446→    const SNAP_THRESHOLD = 8;
   447→    const cx = width / 2;
   448→    const cy = height / 2;
   449→
   450→    // Center guides
   451→    let xGuide: number | undefined;
   452→    let yGuide: number | undefined;
   453→
   454→    // Horizontal center snap
   455→    if (Math.abs(x - cx) < SNAP_THRESHOLD) xGuide = cx;
   456→    // Left margin snap
   457→    if (Math.abs(x - 50) < SNAP_THRESHOLD) xGuide = 50;
   458→    // Right margin snap
   459→    if (Math.abs(x - (width - 50)) < SNAP_THRESHOLD) xGuide = width - 50;
   460→    // Quarter snaps
   461→    if (Math.abs(x - width / 4) < SNAP_THRESHOLD) xGuide = width / 4;
   462→    if (Math.abs(x - (3 * width / 4)) < SNAP_THRESHOLD) xGuide = 3 * width / 4;
   463→
   464→    // Vertical center snap
   465→    if (Math.abs(y - cy) < SNAP_THRESHOLD) yGuide = cy;
   466→    // Top margin snap
   467→    if (Math.abs(y - 50) < SNAP_THRESHOLD) yGuide = 50;
   468→    // Bottom margin snap
   469→    if (Math.abs(y - (height - 50)) < SNAP_THRESHOLD) yGuide = height - 50;
   470→
   471→    // Element-to-element snapping: check against other text elements
   472→    const allTextPositions = Object.entries(textElements).filter(([k]) => {
   473→      if (selectedElement?.kind === 'text' && k === selectedElement.key) return false;
   474→      return true;
   475→    });
   476→    for (const [, pos] of allTextPositions) {
   477→      if (Math.abs(x - pos.x) < SNAP_THRESHOLD && !xGuide) xGuide = pos.x;
   478→      if (Math.abs(y - pos.y) < SNAP_THRESHOLD && !yGuide) yGuide = pos.y;
   479→    }
   480→
   481→    // Element-to-element snapping: check against other logos
   482→    const allLogoPositions = Object.entries(logoElements).filter(([k]) => {
   483→      if (selectedElement?.kind === 'logo' && k === selectedElement.key) return false;
   484→      return true;
