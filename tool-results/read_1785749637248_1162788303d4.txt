    43→function timeAgo(date: string): string {
    44→  const now = new Date();
    45→  const then = new Date(date);
    46→  const seconds = Math.floor((now.getTime() - then.getTime()) / 1000);
    47→  if (seconds < 60) return 'just now';
    48→  const minutes = Math.floor(seconds / 60);
    49→  if (minutes < 60) return `${minutes}m ago`;
    50→  const hours = Math.floor(minutes / 60);
    51→  if (hours < 24) return `${hours}h ago`;
    52→  const days = Math.floor(hours / 24);
    53→  if (days < 2) return 'yesterday';
    54→  return `${days}d ago`;
    55→}
    56→
    57→function getDaysUntil(dateStr: string): number {
    58→  const now = new Date();
    59→  now.setHours(0, 0, 0, 0);
    60→  const target = new Date(dateStr);
    61→  target.setHours(0, 0, 0, 0);
    62→  return Math.ceil((target.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
    63→}

    65→function getGreeting(): string {
