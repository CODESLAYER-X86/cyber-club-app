    41→    41→function timeAgo(date: string): string {
    42→    42→  const now = new Date();
    43→    43→  const then = new Date(date);
    44→    44→  const seconds = Math.floor((now.getTime() - then.getTime()) / 1000);
    45→    45→  if (seconds < 60) return 'just now';
    46→    46→  const minutes = Math.floor(seconds / 60);
    47→    47→  if (minutes < 60) return `${minutes}m ago`;
    48→    48→  const hours = Math.floor(minutes / 60);
    49→    49→  if (hours < 24) return `${hours}h ago`;
    50→    50→  const days = Math.floor(hours / 24);
    51→    51→  if (days < 2) return 'yesterday';
    52→    52→  return `${days}d ago`;
    53→    53→}
    54→    54→
    55→    55→function getDaysUntil(dateStr: string): number {
    56→    56→  const now = new Date();
    57→    57→  now.setHours(0, 0, 0, 0);
    58→    58→  const target = new Date(dateStr);
    59→    59→  target.setHours(0, 0, 0, 0);
    60→    60→  return Math.ceil((target.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
    61→    61→}
