    19→type EventDetailData = Omit<Event, 'registrations'> & {
    20→  creator?: Pick<UserType, 'id' | 'name' | 'email' | 'avatar' | 'role'>;
    21→  verifier?: Pick<UserType, 'id' | 'name' | 'email' | 'avatar' | 'role'>;
    22→  registrations?: (EventRegistration & {

    24→  })[];
    25→  _count?: { registrations: number };
    26→};
    27→const parsePaymentConfig = (paymentConfig?: string | null) => {
    28→  if (!paymentConfig) return null;
    29→  try {
    30→    return JSON.parse(paymentConfig);
    31→  } catch {
    32→    return null;
    33→  }
    34→};

    36→  const { currentUser, selectedEventId, setCurrentView, setEditingEventId } = useAppStore();
    37→  const [event, setEvent] = useState<EventDetailData | null>(null);
    38→  const [loading, setLoading] = useState(true);
    39→  const [registering, setRegistering] = useState(false);
    40→  const [transactionId, setTransactionId] = useState('');
    41→  const [paymentMethod, setPaymentMethod] = useState('BKASH');
    42→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    43→  const [userRegistration, setUserRegistration] = useState<EventRegistration | null>(null);
    44→  const [showRegistrants, setShowRegistrants] = useState(false);
    45→  const [shareMsg, setShareMsg] = useState<string | null>(null);
    46→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
    47→  const [deleting, setDeleting] = useState(false);
    48→  const [updatingCertUserId, setUpdatingCertUserId] = useState<string | null>(null);
    49→  const [updatingRegId, setUpdatingRegId] = useState<string | null>(null);
    50→
    51→  // Enhanced certificate states
    52→  const [prefName, setPrefName] = useState('');
    53→  const [studId, setStudId] = useState('');
    54→  const [dept, setDept] = useState('');
    55→  const [inst, setInst] = useState('');
    56→  const [savingName, setSavingName] = useState(false);
    57→  const [certificateStatus, setCertificateStatus] = useState<string>('REGISTERED');
    58→  const [updatingAttendanceId, setUpdatingAttendanceId] = useState<string | null>(null);
    59→
    60→  const loadEvent = useCallback(async () => {
    61→    if (!selectedEventId) return;
    62→    setLoading(true);
    63→    try {
    64→      const res = await fetch(`/api/events/${selectedEventId}`);
    65→      const data = await res.json();
    66→      if (data.success) {
    67→        setEvent(data.data.event);
    68→        // Check if current user is already registered
    69→        if (currentUser && data.data.event.registrations) {
    70→          const reg = data.data.event.registrations.find(
    71→            (r: EventRegistration) => r.userId === currentUser.id
    72→          );
    73→          setUserRegistration(reg || null);
    74→
    75→          // Fetch certificate status if registered
    76→          try {
    77→            const certRes = await fetch(`/api/certificates?userId=${currentUser.id}`);
    78→            const certData = await certRes.json();
    79→            if (certData.success && certData.data.certificates) {
    80→              const userCert = certData.data.certificates.find((c: any) => c.eventId === selectedEventId);
    81→              if (userCert) {
    82→                setCertificateStatus(userCert.status);
    83→              }
    84→            }
    85→          } catch (e) {
    86→            console.error("Failed to load certificate status:", e);
    87→          }
    88→        }
    89→      }
    90→    } catch (e) {
    91→      console.error(e);
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, [selectedEventId, currentUser]);
    96→
    97→  useEffect(() => {
    98→    loadEvent();
    99→  }, [loadEvent]);
   100→
   101→  // Sync preferred name fields when registration loads
   102→  useEffect(() => {
   103→    if (userRegistration) {
   104→      setPrefName(userRegistration.preferredName || '');
   105→      setStudId(userRegistration.studentId || currentUser?.studentId || '');
   106→      setDept(userRegistration.department || currentUser?.department || '');
   107→      setInst(userRegistration.institution || '');
   108→    }
   109→  }, [userRegistration, currentUser]);
   110→
   111→  const handleSaveName = async () => {
   112→    if (!userRegistration || !currentUser || !event) return;
   113→    setSavingName(true);
   114→    try {
   115→      const res = await fetch(`/api/events/${event.id}/registrations/${userRegistration.id}/name`, {
   116→        method: 'PATCH',
   117→        headers: { 'Content-Type': 'application/json' },
   118→        body: JSON.stringify({
   119→          preferredName: prefName,
   120→          studentId: studId,
   121→          department: dept,
   122→          institution: inst,
   123→          requestingUserId: currentUser.id,
   124→          requestingUserRole: currentUser.role,
   125→        }),
   126→      });
   127→      const data = await res.json();
   128→      if (data.success) {
   129→        toast({ title: 'Information Saved', description: 'Your certificate details have been updated.' });
   130→        loadEvent();
   131→      } else {
   132→        toast({ title: 'Failed to Save', description: data.error || 'Could not update information', variant: 'destructive' });
   133→      }
   134→    } catch {
   135→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   136→    } finally {
   137→      setSavingName(false);
   138→    }
   139→  };
   140→
   141→  const handleMarkAttendance = async (userId: string, status: 'PRESENT' | 'ABSENT' | 'LATE') => {
   142→    if (!event || !currentUser) return;
   143→    setUpdatingAttendanceId(userId);
   144→    try {
   145→      const r = await fetch(`/api/events/${event.id}/attendance`, {
   146→        method: 'POST',
   147→        headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify({
   149→          userId,
   150→          status,
   151→          verifierRole: currentUser.role,
   152→          verifierId: currentUser.id,
   153→        }),
   154→      });
   155→      const d = await r.json();
   156→      if (d.success) {
   157→        toast({ title: 'Attendance Marked', description: `Attendance marked as ${status.toLowerCase()}.` });
   158→        loadEvent();
   159→      } else {
   160→        toast({ title: 'Error', description: d.error || 'Failed to update attendance', variant: 'destructive' });
   161→      }
   162→    } catch {
   163→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   164→    } finally {
   165→      setUpdatingAttendanceId(null);
   166→    }
   167→  };
   168→
   169→  const handleUpdateCertType = async (userId: string, type: string) => {
   170→    if (!event || !currentUser) return;
   171→    setUpdatingCertUserId(userId);
   172→    try {
   173→      const res = await fetch('/api/certificates', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({
   177→          userId,
   178→          eventId: event.id,
   179→          type,
   180→        }),
   181→      });
   182→      const d = await res.json();
   183→      if (d.success) {
   184→        toast({ title: 'Certificate Updated', description: `Assigned as ${CERTIFICATE_TYPE_LABELS[type as CertificateType]}.` });
   185→        loadEvent();
   186→      } else {
   187→        toast({ title: 'Failed to update', description: d.error || 'Please try again', variant: 'destructive' });
   188→      }
   189→    } catch {
   190→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   191→    } finally {
   192→      setUpdatingCertUserId(null);
   193→    }
   194→  };
   195→
   196→  const handleRegister = async () => {
   197→    if (!event || !currentUser) return;
   198→    setRegistering(true);
   199→    setMessage(null);
   200→    try {
   201→      const res = await fetch(`/api/events/${event.id}/register`, {
   202→        method: 'POST',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify({
   205→          userId: currentUser.id,
   206→          transactionId: event.fee > 0 ? transactionId : undefined,
   207→          paymentMethod: event.fee > 0 ? paymentMethod : undefined,
   208→        }),
   209→      });
   210→      const data = await res.json();

   212→        const reg = data.data.registration;
   213→        const isPaid = event.fee > 0;
   214→        const successMsg = isPaid

   216→          : 'Successfully registered! Your registration has been approved.';
   217→        setMessage({ type: 'success', text: successMsg });
   218→        setUserRegistration(reg || { status: isPaid ? 'PENDING' : 'APPROVED' });
   219→        // Reload event data to get updated registrations list and seat count
   220→        loadEvent();
   221→      } else {
   222→        setMessage({ type: 'error', text: data.error || 'Registration failed' });
   223→      }
   224→    } catch {
   225→      setMessage({ type: 'error', text: 'Network error' });
   226→    } finally {
   227→      setRegistering(false);
   228→    }
   229→  };
   230→
   231→  const handleShare = async () => {
   232→    const url = `${window.location.origin}?event=${event?.id}`;
   233→    try {
   234→      await navigator.clipboard.writeText(url);
   235→      setShareMsg('Link copied!');
   236→    } catch {
   237→      setShareMsg('Could not copy link');
   238→    }
   239→    setTimeout(() => setShareMsg(null), 2000);
   240→  };
   241→
   242→  const handleEdit = () => {
   243→    setEditingEventId(event?.id || null);
   244→    setCurrentView('create-event');
   245→  };
   246→

   249→    setDeleting(true);
   250→    try {
   251→      const res = await fetch(`/api/events/${event.id}`, {
   252→        method: 'DELETE',
   253→        headers: { 'Content-Type': 'application/json' },
   254→        body: JSON.stringify({ role: currentUser.role }),
   255→      });
   256→      const data = await res.json();
   257→      if (data.success) {
   258→        toast({ title: 'Event deleted', description: `"${event.title}" has been deleted.` });
   259→        setCurrentView('events');
   260→      } else {
   261→        toast({ title: 'Delete failed', description: data.error || 'Could not delete event', variant: 'destructive' });
   262→      }
   263→    } catch {
   264→      toast({ title: 'Delete failed', description: 'Something went wrong', variant: 'destructive' });
   265→    } finally {
   266→      setDeleting(false);
   267→      setDeleteDialogOpen(false);
   268→    }
   269→  };
   270→
   271→  if (loading) {
   272→    return (
   273→      <div className="space-y-4">
   274→        <div className="h-8 w-48 animate-pulse rounded bg-white/5" />
   275→        <div className="h-64 animate-pulse rounded-xl bg-white/5" />
   276→      </div>
   277→    );
   278→  }
   279→  if (!event) return <div className="py-16 text-center text-gray-500">Event not found</div>;
   280→
   281→  const seatPercent = event.maxSeats ? Math.min((event.currentSeats / event.maxSeats) * 100, 100) : 0;
   282→  const isFull = event.maxSeats ? event.currentSeats >= event.maxSeats : false;
   283→  const isMember = currentUser?.membershipStatus === 'ACTIVE';
   284→  // PUBLIC events: anyone logged in can register
   285→  // MEMBER_ONLY events: only active members can register
   286→  // PAID events: anyone logged in can register (payment required)
   287→  // LIMITED events: anyone logged in can register (subject to seat availability)
   288→  const canRegisterForEventType = event.type === 'MEMBER_ONLY' ? isMember : !!currentUser;
   289→  const canRegister = canRegisterForEventType && !isFull && event.status === 'UPCOMING' && !userRegistration;
   290→  const isAdmin = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'VP', 'GS'].includes(currentUser.role);
   291→  const canApproveReg = currentUser && (
   292→    ['PLATFORM_ADMIN', 'PRESIDENT', 'VP', 'GS', 'VERIFIER'].includes(currentUser.role) ||
   293→    event.verifierId === currentUser.id ||
   294→    event.createdBy === currentUser.id
   295→  );
   296→  const canEdit = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'MEDIA', 'GS'].includes(currentUser.role);
   297→  const canDelete = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'MEDIA', 'VP', 'GS'].includes(currentUser.role);
   298→  const registrationCount = event._count?.registrations ?? event.registrations?.length ?? event.currentSeats;
   299→  const paymentConfig = parsePaymentConfig(event.paymentConfig);
   300→  const paymentFee = paymentConfig?.feeAmount ?? event.fee ?? 0;
   301→  const paymentMethods = [
   302→    { key: 'bkashNumber', label: 'bKash', value: paymentConfig?.bkashNumber },
   303→    { key: 'nagadNumber', label: 'Nagad', value: paymentConfig?.nagadNumber },
   304→    { key: 'rocketNumber', label: 'Rocket', value: paymentConfig?.rocketNumber },
   305→    { key: 'bankAccount', label: 'Bank Transfer', value: paymentConfig?.bankAccount },
   306→  ].filter((method) => method.value);
   307→

   310→      {/* Back Button */}

   312→        <Button variant="ghost" onClick={() => setCurrentView('events')} className="text-gray-400 hover:text-white">
   313→          <ArrowLeft className="mr-2 h-4 w-4" /> Back to Events
   314→        </Button>
   315→        <div className="flex items-center gap-2">
   316→          {/* Share Button */}
   317→          <div className="relative">
   318→            <Button variant="outline" size="sm" onClick={handleShare} className="border-white/10 bg-white/5 text-gray-400 hover:text-white hover:border-emerald-500/30">
   319→              <Share2 className="mr-2 h-4 w-4" /> Share
   320→            </Button>
   321→            {shareMsg && (
   322→              <motion.div
   323→                initial={{ opacity: 0, y: 5 }}
   324→                animate={{ opacity: 1, y: 0 }}
   325→                exit={{ opacity: 0 }}
   326→                className="absolute -bottom-8 right-0 whitespace-nowrap rounded bg-emerald-600 px-2 py-1 text-xs text-white"
   327→              >
   328→                {shareMsg}
   329→              </motion.div>
   330→            )}
   331→          </div>
   332→          {/* Edit Button */}
   333→          {canEdit && (
   334→            <div className="flex gap-2">
   335→              <Button variant="outline" size="sm" onClick={() => setCurrentView('certificate-designer')} className="border-emerald-500/20 text-emerald-400 hover:bg-emerald-500/10 hover:text-white">
   336→                <Award className="mr-2 h-4 w-4" /> Design Certificate
   337→              </Button>
   338→              <Button variant="outline" size="sm" onClick={handleEdit} className="border-white/10 bg-white/5 text-gray-400 hover:text-white hover:border-emerald-500/30">
   339→                <Pencil className="mr-2 h-4 w-4" /> Edit Event
   340→              </Button>
   341→            </div>
   342→          )}
   343→          {/* Delete Button */}
   344→          {canDelete && (
   345→            <Button variant="outline" size="sm" onClick={() => setDeleteDialogOpen(true)} className="border-red-500/20 bg-red-500/5 text-red-400 hover:text-white hover:bg-red-500/20 hover:border-red-500/40">
   346→              <Trash2 className="mr-2 h-4 w-4" /> Delete
   347→            </Button>
   348→          )}
   349→          {/* Mark as Completed Button */}
   350→          {isAdmin && (event.status === 'UPCOMING' || event.status === 'ONGOING') && (
   351→            <Button variant="outline" size="sm" onClick={async () => {
   352→              try {
   353→                const res = await fetch(`/api/events/${event.id}`, {
   354→                  method: 'PATCH',
   355→                  headers: { 'Content-Type': 'application/json' },
   356→                  body: JSON.stringify({ status: 'COMPLETED' }),
   357→                });
   358→                const d = await res.json();
   359→                if (d.success) {
   360→                  toast({ title: 'Event completed', description: 'Event marked as completed. Certificates can now be issued.' });
   361→                  loadEvent();
   362→                } else {
   363→                  toast({ title: 'Failed', description: d.error || 'Could not update event', variant: 'destructive' });
   364→                }
   365→              } catch {
   366→                toast({ title: 'Failed', description: 'Network error', variant: 'destructive' });
   367→              }
   368→            }} className="border-cyan-500/20 bg-cyan-500/5 text-cyan-400 hover:text-white hover:bg-cyan-500/20 hover:border-cyan-500/40">
   369→              <Flag className="mr-2 h-4 w-4" /> Mark Completed
   370→            </Button>

   372→        </div>

   374→
   375→      {/* Event Detail Card */}
   376→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>

   381→              <Badge variant="outline" className="border-white/10 text-gray-400">{EVENT_TYPE_LABELS[event.type]}</Badge>
   382→              <Badge variant="outline" className="border-white/10 text-gray-400">{EVENT_CATEGORY_LABELS[event.category]}</Badge>

   389→            <div className="mt-8 grid gap-6 sm:grid-cols-2">

   391→                <div className="flex items-center gap-3">
   392→                  <Calendar className="h-5 w-5 text-emerald-400" />

   406→                  <MapPin className="h-5 w-5 text-amber-400" />
   407→                  <div>
   408→                    <p className="text-sm font-medium text-white">Venue</p>
   409→                    <p className="text-xs text-gray-500">{event.venue}</p>
   410→                  </div>
   411→                </div>
   412→              </div>
   413→              <div className="space-y-4">
   414→                <div className="flex items-center gap-3">

   420→                </div>
   421→                <div className="flex items-center gap-3">
   422→                  <Users className="h-5 w-5 text-cyan-400" />
   423→                  <div>

   427→                </div>
   428→                {event.requiresAssessment && (
   429→                  <div className="flex items-center gap-3">
   430→                    <Award className="h-5 w-5 text-amber-400" />

   432→                      <p className="text-sm font-medium text-white">Passing Score</p>

   435→                  </div>

   437→              </div>

   439→
   440→            {/* Organizer & Verifier Info */}
   441→            <Separator className="my-6 bg-white/5" />
   442→            <div className="grid gap-4 sm:grid-cols-2">

   444→                <div className="flex items-center gap-3">
   445→                  <div className="flex h-10 w-10 items-center justify-center rounded-full border border-emerald-500/20 bg-emerald-500/10">

   467→            </div>
   468→
   469→            {/* Seat Progress */}
   470→            {event.maxSeats && (
   471→              <div className="mt-6">
