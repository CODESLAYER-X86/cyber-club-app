   430→   430→
   431→   431→  const svgPointFromEvent = (event: PointerEvent | React.PointerEvent) => {
   432→   432→    const svg = svgRef.current;
   433→   433→    if (!svg) return { x: 0, y: 0 };
   434→   434→    const point = svg.createSVGPoint();
   435→   435→    point.x = event.clientX;
   436→   436→    point.y = event.clientY;
   437→   437→    const ctm = svg.getScreenCTM();
   438→   438→    if (!ctm) return { x: 0, y: 0 };
   439→   439→    const transformed = point.matrixTransform(ctm.inverse());
   440→   440→    return { x: transformed.x, y: transformed.y };
   441→   441→  };

   443→   443→  /* ─── Enhanced Alignment Guides ─── */
   444→   444→
   445→   445→  const snapToGuides = (x: number, y: number, elementWidth?: number, elementHeight?: number) => {
   446→   446→    const SNAP_THRESHOLD = 8;
   447→   447→    const cx = width / 2;
   448→   448→    const cy = height / 2;
   449→   449→
   450→   450→    // Center guides
   451→   451→    let xGuide: number | undefined;
   452→   452→    let yGuide: number | undefined;
   453→   453→
   454→   454→    // Horizontal center snap
   455→   455→    if (Math.abs(x - cx) < SNAP_THRESHOLD) xGuide = cx;
   456→   456→    // Left margin snap
   457→   457→    if (Math.abs(x - 50) < SNAP_THRESHOLD) xGuide = 50;
   458→   458→    // Right margin snap
   459→   459→    if (Math.abs(x - (width - 50)) < SNAP_THRESHOLD) xGuide = width - 50;
   460→   460→    // Quarter snaps
   461→   461→    if (Math.abs(x - width / 4) < SNAP_THRESHOLD) xGuide = width / 4;
   462→   462→    if (Math.abs(x - (3 * width / 4)) < SNAP_THRESHOLD) xGuide = 3 * width / 4;
   463→   463→
   464→   464→    // Vertical center snap
   465→   465→    if (Math.abs(y - cy) < SNAP_THRESHOLD) yGuide = cy;
   466→   466→    // Top margin snap
   467→   467→    if (Math.abs(y - 50) < SNAP_THRESHOLD) yGuide = 50;
   468→   468→    // Bottom margin snap
   469→   469→    if (Math.abs(y - (height - 50)) < SNAP_THRESHOLD) yGuide = height - 50;
   470→   470→
   471→   471→    // Element-to-element snapping: check against other text elements
   472→   472→    const allTextPositions = Object.entries(textElements).filter(([k]) => {
   473→   473→      if (selectedElement?.kind === 'text' && k === selectedElement.key) return false;
   474→   474→      return true;
   475→   475→    });
   476→   476→    for (const [, pos] of allTextPositions) {
   477→   477→      if (Math.abs(x - pos.x) < SNAP_THRESHOLD && !xGuide) xGuide = pos.x;
   478→   478→      if (Math.abs(y - pos.y) < SNAP_THRESHOLD && !yGuide) yGuide = pos.y;
   479→   479→    }
   480→   480→
   481→   481→    // Element-to-element snapping: check against other logos
   482→   482→    const allLogoPositions = Object.entries(logoElements).filter(([k]) => {
   483→   483→      if (selectedElement?.kind === 'logo' && k === selectedElement.key) return false;
   484→   484→      return true;
   485→   485→    });
   486→   486→    for (const [, pos] of allLogoPositions) {
   487→   487→      if (Math.abs(x - pos.x) < SNAP_THRESHOLD && !xGuide) xGuide = pos.x;
   488→   488→      if (Math.abs(y - pos.y) < SNAP_THRESHOLD && !yGuide) yGuide = pos.y;
   489→   489→      // Right edge snap
   490→   490→      if (elementWidth && Math.abs(x - (pos.x + pos.width)) < SNAP_THRESHOLD && !xGuide) xGuide = pos.x + pos.width;
   491→   491→      // Bottom edge snap
   492→   492→      if (elementHeight && Math.abs(y - (pos.y + pos.height)) < SNAP_THRESHOLD && !yGuide) yGuide = pos.y + pos.height;
   493→   493→    }
   494→   494→
   495→   495→    return {
   496→   496→      x: xGuide ?? x,
   497→   497→      y: yGuide ?? y,
   498→   498→      guides: { vertical: xGuide, horizontal: yGuide },
   499→   499→    };
   500→   500→  };
