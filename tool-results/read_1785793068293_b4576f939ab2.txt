    83→function timeAgo(dateStr: string): string {
    84→  const now = new Date();
    85→  const date = new Date(dateStr);
    86→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000);
    87→
    88→  if (seconds < 60) return 'just now';
    89→  const minutes = Math.floor(seconds / 60);
    90→  if (minutes < 60) return `${minutes}m ago`;
    91→  const hours = Math.floor(minutes / 60);
    92→  if (hours < 24) return `${hours}h ago`;
    93→  const days = Math.floor(hours / 24);
    94→  if (days === 1) return 'yesterday';
    95→  if (days < 7) return `${days}d ago`;
    96→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
    97→  return date.toLocaleDateString();
    98→}
