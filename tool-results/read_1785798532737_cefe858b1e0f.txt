    39→function parsePaymentConfig(paymentConfig?: string | null) {
    40→  if (!paymentConfig) return null;
    41→  try {
    42→    return JSON.parse(paymentConfig);
    43→  } catch {
    44→    return null;
    45→  }

    91→            description: e.description || '',

    94→            startDate: toLocalDatetime(e.startDate),
    95→            endDate: toLocalDatetime(e.endDate),
    96→            venue: e.venue || '',
    97→            fee: String(e.fee ?? 0),
    98→            maxSeats: e.maxSeats ? String(e.maxSeats) : '',

   100→            passingScore: e.passingScore ? String(e.passingScore) : '60',

   102→            bkashNumber: pc?.bkashNumber || '',
   103→            nagadNumber: pc?.nagadNumber || '',
   104→            rocketNumber: pc?.rocketNumber || '',
