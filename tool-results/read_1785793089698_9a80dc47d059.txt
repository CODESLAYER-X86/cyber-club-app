     1→'use client';
     2→

     4→import { motion, AnimatePresence } from 'framer-motion';

     6→import { useAppStore } from '@/store/use-app-store';
     7→import type { Event, EventRegistration, User as UserType } from '@/types';
     8→import { EVENT_TYPE_LABELS, EVENT_CATEGORY_LABELS, ROLE_LABELS, CERTIFICATE_TYPE_LABELS, CertificateType } from '@/types';
     9→import { EventBadge, RegistrationBadge } from '@/components/shared/status-badge';
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    11→import { Button } from '@/components/ui/button';
    12→import { Badge } from '@/components/ui/badge';
    13→import { Input } from '@/components/ui/input';
    14→import { Progress } from '@/components/ui/progress';
    15→import { Separator } from '@/components/ui/separator';
    16→import { AlertDialog, AlertDialogContent, AlertDialogHeader, AlertDialogTitle, AlertDialogDescription, AlertDialogFooter, AlertDialogCancel, AlertDialogAction } from '@/components/ui/alert-dialog';
    17→import { toast } from '@/hooks/use-toast';
    18→
    19→type EventDetailData = Omit<Event, 'registrations'> & {
    20→  creator?: Pick<UserType, 'id' | 'name' | 'email' | 'avatar' | 'role'>;
    21→  verifier?: Pick<UserType, 'id' | 'name' | 'email' | 'avatar' | 'role'>;
    22→  registrations?: (EventRegistration & {
    23→    user?: Pick<UserType, 'id' | 'name' | 'email' | 'avatar' | 'role' | 'membershipStatus'>;
    24→  })[];
    25→  _count?: { registrations: number };
    26→};
    27→const parsePaymentConfig = (paymentConfig?: string | null) => {
    28→  if (!paymentConfig) return null;
    29→  try {
    30→    return JSON.parse(paymentConfig);
    31→  } catch {
    32→    return null;
    33→  }
    34→};
    35→export function EventDetailPage() {
    36→  const { currentUser, selectedEventId, setCurrentView } = useAppStore();
    37→  const [event, setEvent] = useState<EventDetailData | null>(null);
    38→  const [loading, setLoading] = useState(true);
    39→  const [registering, setRegistering] = useState(false);
    40→  const [transactionId, setTransactionId] = useState('');
    41→  const [paymentMethod, setPaymentMethod] = useState('BKASH');
    42→  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);
    43→  const [userRegistration, setUserRegistration] = useState<EventRegistration | null>(null);
    44→  const [showRegistrants, setShowRegistrants] = useState(false);
    45→  const [shareMsg, setShareMsg] = useState<string | null>(null);
    46→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
    47→  const [deleting, setDeleting] = useState(false);
    48→  const [updatingCertUserId, setUpdatingCertUserId] = useState<string | null>(null);
    49→  const [updatingRegId, setUpdatingRegId] = useState<string | null>(null);
    50→
    51→  // Enhanced certificate states
    52→  const [prefName, setPrefName] = useState('');
    53→  const [studId, setStudId] = useState('');
    54→  const [dept, setDept] = useState('');
    55→  const [inst, setInst] = useState('');
    56→  const [savingName, setSavingName] = useState(false);
    57→  const [certificateStatus, setCertificateStatus] = useState<string>('REGISTERED');
    58→  const [updatingAttendanceId, setUpdatingAttendanceId] = useState<string | null>(null);
    59→
    60→  const loadEvent = useCallback(async () => {
    61→    if (!selectedEventId) return;
    62→    setLoading(true);
    63→    try {
    64→      const res = await fetch(`/api/events/${selectedEventId}`);
    65→      const data = await res.json();
    66→      if (data.success) {
    67→        setEvent(data.data.event);
    68→        // Check if current user is already registered
    69→        if (currentUser && data.data.event.registrations) {
    70→          const reg = data.data.event.registrations.find(
    71→            (r: EventRegistration) => r.userId === currentUser.id
    72→          );
    73→          setUserRegistration(reg || null);
    74→
    75→          // Fetch certificate status if registered
    76→          try {
    77→            const certRes = await fetch(`/api/certificates?userId=${currentUser.id}`);
    78→            const certData = await certRes.json();
    79→            if (certData.success && certData.data.certificates) {
    80→              const userCert = certData.data.certificates.find((c: any) => c.eventId === selectedEventId);
    81→              if (userCert) {
    82→                setCertificateStatus(userCert.status);
    83→              }
    84→            }
    85→          } catch (e) {
    86→            console.error("Failed to load certificate status:", e);
    87→          }
    88→        }
    89→      }
    90→    } catch (e) {
    91→      console.error(e);
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, [selectedEventId, currentUser]);
    96→
    97→  useEffect(() => {
    98→    loadEvent();
    99→  }, [loadEvent]);
   100→
   101→  // Sync preferred name fields when registration loads
   102→  useEffect(() => {
   103→    if (userRegistration) {
   104→      setPrefName(userRegistration.preferredName || '');
   105→      setStudId(userRegistration.studentId || currentUser?.studentId || '');
   106→      setDept(userRegistration.department || currentUser?.department || '');
   107→      setInst(userRegistration.institution || '');
   108→    }
   109→  }, [userRegistration, currentUser]);
   110→
   111→  const handleSaveName = async () => {
   112→    if (!userRegistration || !currentUser || !event) return;
   113→    setSavingName(true);
   114→    try {
   115→      const res = await fetch(`/api/events/${event.id}/registrations/${userRegistration.id}/name`, {
   116→        method: 'PATCH',
   117→        headers: { 'Content-Type': 'application/json' },
   118→        body: JSON.stringify({
   119→          preferredName: prefName,
   120→          studentId: studId,
   121→          department: dept,
   122→          institution: inst,
   123→          requestingUserId: currentUser.id,
   124→          requestingUserRole: currentUser.role,
   125→        }),
   126→      });
   127→      const data = await res.json();
   128→      if (data.success) {
   129→        toast({ title: 'Information Saved', description: 'Your certificate details have been updated.' });
   130→        loadEvent();
   131→      } else {
   132→        toast({ title: 'Failed to Save', description: data.error || 'Could not update information', variant: 'destructive' });
   133→      }
   134→    } catch {
   135→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   136→    } finally {
   137→      setSavingName(false);
   138→    }
   139→  };
   140→
   141→  const handleMarkAttendance = async (userId: string, status: 'PRESENT' | 'ABSENT' | 'LATE') => {
   142→    if (!event || !currentUser) return;
   143→    setUpdatingAttendanceId(userId);
   144→    try {
   145→      const r = await fetch(`/api/events/${event.id}/attendance`, {
   146→        method: 'POST',
   147→        headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify({
   149→          userId,
   150→          status,
   151→          verifierRole: currentUser.role,
   152→          verifierId: currentUser.id,
   153→        }),
   154→      });
   155→      const d = await r.json();
   156→      if (d.success) {
   157→        toast({ title: 'Attendance Marked', description: `Attendance marked as ${status.toLowerCase()}.` });
   158→        loadEvent();
   159→      } else {
   160→        toast({ title: 'Error', description: d.error || 'Failed to update attendance', variant: 'destructive' });
   161→      }
   162→    } catch {
   163→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   164→    } finally {
   165→      setUpdatingAttendanceId(null);
   166→    }
   167→  };
   168→
   169→  const handleUpdateCertType = async (userId: string, type: string) => {
   170→    if (!event || !currentUser) return;
   171→    setUpdatingCertUserId(userId);
   172→    try {
   173→      const res = await fetch('/api/certificates', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({
   177→          userId,
   178→          eventId: event.id,
   179→          type,
   180→        }),
   181→      });
   182→      const d = await res.json();
   183→      if (d.success) {
   184→        toast({ title: 'Certificate Updated', description: `Assigned as ${CERTIFICATE_TYPE_LABELS[type as CertificateType]}.` });
   185→        loadEvent();
   186→      } else {
   187→        toast({ title: 'Failed to update', description: d.error || 'Please try again', variant: 'destructive' });
   188→      }
   189→    } catch {
   190→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' });
   191→    } finally {
   192→      setUpdatingCertUserId(null);
   193→    }
   194→  };
   195→
   196→  const handleRegister = async () => {
   197→    if (!event || !currentUser) return;
   198→    setRegistering(true);
   199→    setMessage(null);
   200→    try {
   201→      const res = await fetch(`/api/events/${event.id}/register`, {
   202→        method: 'POST',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify({
   205→          userId: currentUser.id,
   206→          transactionId: event.fee > 0 ? transactionId : undefined,
   207→          paymentMethod: event.fee > 0 ? paymentMethod : undefined,
   208→        }),
   209→      });
   210→      const data = await res.json();
   211→      if (data.success) {
   212→        const reg = data.data.registration;
   213→        const isPaid = event.fee > 0;
   214→        const successMsg = isPaid
   215→          ? 'Successfully registered! Your payment is pending verification. You will be approved once payment is confirmed.'
   216→          : 'Successfully registered! Your registration has been approved.';
   217→        setMessage({ type: 'success', text: successMsg });
   218→        setUserRegistration(reg || { status: isPaid ? 'PENDING' : 'APPROVED' });
   219→        // Reload event data to get updated registrations list and seat count
   220→        loadEvent();
   221→      } else {
   222→        setMessage({ type: 'error', text: data.error || 'Registration failed' });
   223→      }
   224→    } catch {
   225→      setMessage({ type: 'error', text: 'Network error' });
   226→    } finally {
   227→      setRegistering(false);
   228→    }
   229→  };
   230→
   231→  const handleShare = async () => {
   232→    const url = `${window.location.origin}?event=${event?.id}`;
   233→    try {
   234→      await navigator.clipboard.writeText(url);
   235→      setShareMsg('Link copied!');
   236→    } catch {
   237→      setShareMsg('Could not copy link');
   238→    }
   239→    setTimeout(() => setShareMsg(null), 2000);
   240→  };
   241→
   242→  const handleEdit = () => {
   243→    setCurrentView('create-event');
   244→  };
   245→
   246→  const handleDeleteEvent = async () => {
   247→    if (!event || !currentUser) return;
   248→    setDeleting(true);
   249→    try {
   250→      const res = await fetch(`/api/events/${event.id}`, {
   251→        method: 'DELETE',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ role: currentUser.role }),
   254→      });
   255→      const data = await res.json();
   256→      if (data.success) {
   257→        toast({ title: 'Event deleted', description: `"${event.title}" has been deleted.` });
   258→        setCurrentView('events');
   259→      } else {
   260→        toast({ title: 'Delete failed', description: data.error || 'Could not delete event', variant: 'destructive' });
   261→      }
   262→    } catch {
   263→      toast({ title: 'Delete failed', description: 'Something went wrong', variant: 'destructive' });
   264→    } finally {
   265→      setDeleting(false);
   266→      setDeleteDialogOpen(false);
   267→    }
   268→  };
   269→
   270→  if (loading) {
   271→    return (
   272→      <div className="space-y-4">
   273→        <div className="h-8 w-48 animate-pulse rounded bg-white/5" />
   274→        <div className="h-64 animate-pulse rounded-xl bg-white/5" />
   275→      </div>
   276→    );
   277→  }
   278→  if (!event) return <div className="py-16 text-center text-gray-500">Event not found</div>;
   279→
   280→  const seatPercent = event.maxSeats ? Math.min((event.currentSeats / event.maxSeats) * 100, 100) : 0;
   281→  const isFull = event.maxSeats ? event.currentSeats >= event.maxSeats : false;
   282→  const isMember = currentUser?.membershipStatus === 'ACTIVE';
   283→  // PUBLIC events: anyone logged in can register
   284→  // MEMBER_ONLY events: only active members can register
   285→  // PAID events: anyone logged in can register (payment required)
   286→  // LIMITED events: anyone logged in can register (subject to seat availability)
   287→  const canRegisterForEventType = event.type === 'MEMBER_ONLY' ? isMember : !!currentUser;
   288→  const canRegister = canRegisterForEventType && !isFull && event.status === 'UPCOMING' && !userRegistration;
   289→  const isAdmin = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'VP', 'GS'].includes(currentUser.role);
   290→  const canApproveReg = currentUser && (
   291→    ['PLATFORM_ADMIN', 'PRESIDENT', 'VP', 'GS', 'VERIFIER'].includes(currentUser.role) ||
   292→    event.verifierId === currentUser.id ||
   293→    event.createdBy === currentUser.id
   294→  );
   295→  const canEdit = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'MEDIA'].includes(currentUser.role);
   296→  const canDelete = currentUser && ['PLATFORM_ADMIN', 'PRESIDENT', 'MEDIA', 'VP', 'GS'].includes(currentUser.role);
   297→  const registrationCount = event._count?.registrations ?? event.registrations?.length ?? event.currentSeats;
   298→  const paymentConfig = parsePaymentConfig(event.paymentConfig);
   299→  const paymentFee = paymentConfig?.feeAmount ?? event.fee ?? 0;
   300→  const paymentMethods = [
   301→    { key: 'bkashNumber', label: 'bKash', value: paymentConfig?.bkashNumber },
   302→    { key: 'nagadNumber', label: 'Nagad', value: paymentConfig?.nagadNumber },
   303→    { key: 'rocketNumber', label: 'Rocket', value: paymentConfig?.rocketNumber },
   304→    { key: 'bankAccount', label: 'Bank Transfer', value: paymentConfig?.bankAccount },
   305→  ].filter((method) => method.value);
   306→
   307→  return (
   308→    <div className="space-y-6">
   309→      {/* Back Button */}
   310→      <div className="flex items-center justify-between">
   311→        <Button variant="ghost" onClick={() => setCurrentView('events')} className="text-gray-400 hover:text-white">
   312→          <ArrowLeft className="mr-2 h-4 w-4" /> Back to Events
   313→        </Button>
   314→        <div className="flex items-center gap-2">
   315→          {/* Share Button */}
   316→          <div className="relative">
   317→            <Button variant="outline" size="sm" onClick={handleShare} className="border-white/10 bg-white/5 text-gray-400 hover:text-white hover:border-emerald-500/30">
   318→              <Share2 className="mr-2 h-4 w-4" /> Share
   319→            </Button>
   320→            {shareMsg && (
   321→              <motion.div
   322→                initial={{ opacity: 0, y: 5 }}
   323→                animate={{ opacity: 1, y: 0 }}
   324→                exit={{ opacity: 0 }}
   325→                className="absolute -bottom-8 right-0 whitespace-nowrap rounded bg-emerald-600 px-2 py-1 text-xs text-white"
   326→              >
   327→                {shareMsg}
   328→              </motion.div>
   329→            )}
   330→          </div>
   331→          {/* Edit Button */}
   332→          {canEdit && (
   333→            <div className="flex gap-2">
   334→              <Button variant="outline" size="sm" onClick={() => setCurrentView('certificate-designer')} className="border-emerald-500/20 text-emerald-400 hover:bg-emerald-500/10 hover:text-white">
   335→                <Award className="mr-2 h-4 w-4" /> Design Certificate
   336→              </Button>
   337→              <Button variant="outline" size="sm" onClick={handleEdit} className="border-white/10 bg-white/5 text-gray-400 hover:text-white hover:border-emerald-500/30">
   338→                <Pencil className="mr-2 h-4 w-4" /> Edit Event
   339→              </Button>
   340→            </div>
   341→          )}
   342→          {/* Delete Button */}
   343→          {canDelete && (
   344→            <Button variant="outline" size="sm" onClick={() => setDeleteDialogOpen(true)} className="border-red-500/20 bg-red-500/5 text-red-400 hover:text-white hover:bg-red-500/20 hover:border-red-500/40">
   345→              <Trash2 className="mr-2 h-4 w-4" /> Delete
   346→            </Button>
   347→          )}
   348→          {/* Mark as Completed Button */}
   349→          {isAdmin && (event.status === 'UPCOMING' || event.status === 'ONGOING') && (
   350→            <Button variant="outline" size="sm" onClick={async () => {
   351→              try {
   352→                const res = await fetch(`/api/events/${event.id}`, {
   353→                  method: 'PATCH',
   354→                  headers: { 'Content-Type': 'application/json' },
   355→                  body: JSON.stringify({ status: 'COMPLETED' }),
   356→                });
   357→                const d = await res.json();
   358→                if (d.success) {
   359→                  toast({ title: 'Event completed', description: 'Event marked as completed. Certificates can now be issued.' });
   360→                  loadEvent();
   361→                } else {
   362→                  toast({ title: 'Failed', description: d.error || 'Could not update event', variant: 'destructive' });
   363→                }
   364→              } catch {
   365→                toast({ title: 'Failed', description: 'Network error', variant: 'destructive' });
   366→              }
   367→            }} className="border-cyan-500/20 bg-cyan-500/5 text-cyan-400 hover:text-white hover:bg-cyan-500/20 hover:border-cyan-500/40">
   368→              <Flag className="mr-2 h-4 w-4" /> Mark Completed
   369→            </Button>
   370→          )}
   371→        </div>
   372→      </div>
   373→
   374→      {/* Event Detail Card */}
   375→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }}>
   376→        <Card className="border-white/5 bg-[#111]/60 backdrop-blur">
   377→          <CardContent className="pt-6">
   378→            <div className="flex flex-wrap items-center gap-2 mb-4">
   379→              <EventBadge status={event.status} />
   380→              <Badge variant="outline" className="border-white/10 text-gray-400">{EVENT_TYPE_LABELS[event.type]}</Badge>
   381→              <Badge variant="outline" className="border-white/10 text-gray-400">{EVENT_CATEGORY_LABELS[event.category]}</Badge>
   382→              {event.requiresAssessment && <Badge variant="outline" className="border-amber-500/30 text-amber-400">Assessment Required</Badge>}
   383→            </div>
   384→
   385→            <h1 className="text-3xl font-bold text-white">{event.title}</h1>
   386→            <p className="mt-4 text-gray-400 leading-relaxed">{event.description}</p>
   387→
   388→            <div className="mt-8 grid gap-6 sm:grid-cols-2">
   389→              <div className="space-y-4">
   390→                <div className="flex items-center gap-3">
   391→                  <Calendar className="h-5 w-5 text-emerald-400" />
   392→                  <div>
   393→                    <p className="text-sm font-medium text-white">Start Date</p>
   394→                    <p className="text-xs text-gray-500">{new Date(event.startDate).toLocaleString()}</p>
   395→                  </div>
   396→                </div>
   397→                <div className="flex items-center gap-3">
   398→                  <Clock className="h-5 w-5 text-cyan-400" />
   399→                  <div>
   400→                    <p className="text-sm font-medium text-white">End Date</p>
   401→                    <p className="text-xs text-gray-500">{new Date(event.endDate).toLocaleString()}</p>
   402→                  </div>
   403→                </div>
   404→                <div className="flex items-center gap-3">
   405→                  <MapPin className="h-5 w-5 text-amber-400" />
   406→                  <div>
   407→                    <p className="text-sm font-medium text-white">Venue</p>
   408→                    <p className="text-xs text-gray-500">{event.venue}</p>
   409→                  </div>
   410→                </div>
   411→              </div>
   412→              <div className="space-y-4">
   413→                <div className="flex items-center gap-3">
   414→                  <DollarSign className="h-5 w-5 text-emerald-400" />
   415→                  <div>
   416→                    <p className="text-sm font-medium text-white">Fee</p>
   417→                    <p className="text-xs text-gray-500">{event.fee > 0 ? `৳${event.fee}` : 'Free'}</p>
   418→                  </div>
   419→                </div>
   420→                <div className="flex items-center gap-3">
   421→                  <Users className="h-5 w-5 text-cyan-400" />
   422→                  <div>
   423→                    <p className="text-sm font-medium text-white">Registrations</p>
   424→                    <p className="text-xs text-gray-500">{registrationCount} / {event.maxSeats || 'Unlimited'}</p>
   425→                  </div>
   426→                </div>
   427→                {event.requiresAssessment && (
   428→                  <div className="flex items-center gap-3">
   429→                    <Award className="h-5 w-5 text-amber-400" />
   430→                    <div>
   431→                      <p className="text-sm font-medium text-white">Passing Score</p>
   432→                      <p className="text-xs text-gray-500">{event.passingScore}%</p>
   433→                    </div>
   434→                  </div>
   435→                )}
   436→              </div>
   437→            </div>
   438→
   439→            {/* Organizer & Verifier Info */}
   440→            <Separator className="my-6 bg-white/5" />
   441→            <div className="grid gap-4 sm:grid-cols-2">
   442→              {event.creator && (
   443→                <div className="flex items-center gap-3">
   444→                  <div className="flex h-10 w-10 items-center justify-center rounded-full border border-emerald-500/20 bg-emerald-500/10">
   445→                    <User className="h-5 w-5 text-emerald-400" />
   446→                  </div>
   447→                  <div>
   448→                    <p className="text-xs text-gray-500 uppercase tracking-wider">Organizer</p>
   449→                    <p className="text-sm font-medium text-white">{event.creator.name}</p>
   450→                    <p className="text-xs text-gray-500">{ROLE_LABELS[event.creator.role as keyof typeof ROLE_LABELS] || event.creator.role}</p>
   451→                  </div>
   452→                </div>
   453→              )}
   454→              {event.verifier && (
   455→                <div className="flex items-center gap-3">
   456→                  <div className="flex h-10 w-10 items-center justify-center rounded-full border border-cyan-500/20 bg-cyan-500/10">
   457→                    <ShieldCheck className="h-5 w-5 text-cyan-400" />
   458→                  </div>
   459→                  <div>
   460→                    <p className="text-xs text-gray-500 uppercase tracking-wider">Verifier</p>
   461→                    <p className="text-sm font-medium text-white">{event.verifier.name}</p>
   462→                    <p className="text-xs text-gray-500">{ROLE_LABELS[event.verifier.role as keyof typeof ROLE_LABELS] || event.verifier.role}</p>
   463→                  </div>
   464→                </div>
   465→              )}
   466→            </div>
   467→
   468→            {/* Seat Progress */}
   469→            {event.maxSeats && (
   470→              <div className="mt-6">
   471→                <div className="mb-2 flex justify-between text-xs text-gray-500">
   472→                  <span>{event.currentSeats} registered</span>
   473→                  <span>{event.maxSeats} total</span>
