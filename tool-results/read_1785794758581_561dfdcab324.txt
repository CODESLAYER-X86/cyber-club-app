    54→function timeAgo(date: string): string {
    55→  const now = new Date();
    56→  const then = new Date(date);
    57→  const seconds = Math.floor((now.getTime() - then.getTime()) / 1000);
    58→  if (seconds < 60) return 'just now';
    59→  const minutes = Math.floor(seconds / 60);
    60→  if (minutes < 60) return `${minutes}m ago`;
    61→  const hours = Math.floor(minutes / 60);
    62→  if (hours < 24) return `${hours}h ago`;
    63→  const days = Math.floor(hours / 24);
    64→  if (days < 2) return 'yesterday';
    65→  return `${days}d ago`;
    66→}
    67→
    68→function getDaysUntil(dateStr: string): number {
    69→  const now = new Date();
    70→  now.setHours(0, 0, 0, 0);
    71→  const target = new Date(dateStr);
    72→  target.setHours(0, 0, 0, 0);
    73→  return Math.ceil((target.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
    74→}
